    <script>

        // Dados padrão iniciais (Garante que a aba anónima e o telemóvel já abrem com um catálogo preenchido)

        let defaultStore = {

            name: "Lanches da Tina",

            phone: "5531999999999",

            address: "Rua Coronel Pereira Filho, 1343 - São Cristovão, Betim - MG",

            color: "#16a34a", // Verde elegante por defeito

            logo: "https://images.unsplash.com/photo-1555396273-367ea4eb4db5?w=150"

        };



        let defaultCategories = ["Bebidas", "Açaí"];

        

        let defaultProducts = [

            { name: "Coca Cola 350ml", category: "Bebidas", price: "5.99", image: "https://images.unsplash.com/photo-1622483767028-3f66f32aef97?w=150" },

            { name: "Fanta Laranja 350ml", category: "Bebidas", price: "5.99", image: "https://images.unsplash.com/photo-1624552184280-9e9632bbeee9?w=150" },

            { name: "Açaí 300ml", category: "Açaí", price: "14.99", image: "https://images.unsplash.com/photo-1590301157890-4810ed352733?w=150" },

            { name: "Açaí 500ml", category: "Açaí", price: "19.99", image: "https://images.unsplash.com/photo-1590301157890-4810ed352733?w=150" }

        ];



        let storeData = {};

        let categories = [];

        let products = [];

        let cart = [];

        let isAdminVisible = false;



        window.onload = function() {

            const savedStore = localStorage.getItem('storeData');

            const savedCats = localStorage.getItem('categories');

            const savedProducts = localStorage.getItem('products');



            // Se não houver dados salvos (como na aba anónima ou telemóvel novo), usa os padrões criados

            storeData = savedStore ? JSON.parse(savedStore) : defaultStore;

            categories = savedCats ? JSON.parse(savedCats) : defaultCategories;

            products = savedProducts ? JSON.parse(savedProducts) : defaultProducts;



            document.getElementById('storeName').value = storeData.name;

            document.getElementById('storePhone').value = storeData.phone;

            document.getElementById('storeAddress').value = storeData.address;

            document.getElementById('storeColor').value = storeData.color;



            isAdminVisible = false;

            document.getElementById('adminSection').style.display = 'none';

            document.getElementById('btnToggleMode').innerText = "⚙️ Abrir Painel do Comerciante";



            applyStoreTheme();

            renderCategories();

            renderCategorySelect();

            renderCatalog();

            updateCartUI();

        };



        function toggleAdminPanel() {

            isAdminVisible = !isAdminVisible;

            const adminSection = document.getElementById('adminSection');

            const btnToggle = document.getElementById('btnToggleMode');

            

            if (isAdminVisible) {

                adminSection.style.display = 'block';

                btnToggle.innerText = "👁️ Ocultar Painel do Comerciante";

            } else {

                adminSection.style.display = 'none';

                btnToggle.innerText = "⚙️ Abrir Painel do Comerciante";

            }

            renderCatalog();

            updateCartUI(); // Esconde o carrinho se entrar no modo admin

        }



        function updateStoreInfo() {

            storeData.name = document.getElementById('storeName').value || "Meu Negócio";

            storeData.phone = document.getElementById('storePhone').value || "5531999999999";

            storeData.address = document.getElementById('storeAddress').value || "Betim - MG";

            storeData.color = document.getElementById('storeColor').value || "#ea580c";

            

            applyStoreTheme();

            saveData();

        }



        function applyStoreTheme() {

            document.getElementById('mainHeader').style.backgroundColor = storeData.color;

            document.getElementById('btnAddProduct').style.backgroundColor = storeData.color;

            document.getElementById('headerStoreName').innerText = storeData.name;

            document.getElementById('headerStoreAddress').innerText = storeData.address;

            document.getElementById('headerPhone').innerText = "WhatsApp: " + storeData.phone;

            if (storeData.logo) {

                document.getElementById('headerLogo').src = storeData.logo;

            }

        }



        function loadLogoFile(event) {

            const file = event.target.files[0];

            if (file) {

                const reader = new FileReader();

                reader.onload = function(e) {

                    storeData.logo = e.target.result;

                    applyStoreTheme();

                    saveData();

                };

                reader.readAsDataURL(file);

            }

        }



        function addCategory() {

            const catInput = document.getElementById('newCategoryName');

            const catName = catInput.value.trim();

            if (!catName) { alert('Digite o nome da categoria.'); return; }

            if (!categories.includes(catName)) {

                categories.push(catName);

                catInput.value = '';

                saveData();

                renderCategories();

                renderCategorySelect();

            } else {

                alert('Esta categoria já existe.');

            }

        }



        function removeCategory(index) {

            categories.splice(index, 1);

            saveData();

            renderCategories();

            renderCategorySelect();

        }



        function renderCategories() {

            const container = document.getElementById('categoryListTags');

            let html = '';

            categories.forEach((cat, index) => {

                html += `

                    <span class="bg-white border border-blue-200 text-blue-800 text-xs px-3 py-1 rounded-full flex items-center gap-2 shadow-sm font-medium">

                        ${cat}

                        <button onclick="removeCategory(${index})" class="text-red-500 font-bold">×</button>

                    </span>

                `;

            });

            container.innerHTML = html;

        }



        function renderCategorySelect() {

            const select = document.getElementById('prodCategorySelect');

            let html = '<option value="">Selecione a Categoria</option>';

            categories.forEach(cat => { html += `<option value="${cat}">${cat}</option>`; });

            select.innerHTML = html;

        }



        function addProduct() {

            const name = document.getElementById('prodName').value;

            const category = document.getElementById('prodCategorySelect').value;

            const price = document.getElementById('prodPrice').value;

            const fileInput = document.getElementById('prodImageFile');



            if (!name || !price || !category) {

                alert('Preencha o nome, selecione a categoria e informe o preço.');

                return;

            }



            if (fileInput.files && fileInput.files[0]) {

                const reader = new FileReader();

                reader.onload = function(e) { saveProductFinal(name, category, price, e.target.result); };

                reader.readAsDataURL(fileInput.files[0]);

            } else {

                saveProductFinal(name, category, price, 'https://via.placeholder.com/150');

            }

        }



        function saveProductFinal(name, category, price, imageUrl) {

            products.push({ name, category, price: parseFloat(price).toFixed(2), image: imageUrl });

            document.getElementById('prodName').value = '';

            document.getElementById('prodPrice').value = '';

            document.getElementById('prodImageFile').value = '';

            document.getElementById('prodCategorySelect').value = '';

            saveData();

            renderCatalog();

        }



        function removeProduct(index) {

            products.splice(index, 1);

            saveData();

            renderCatalog();

        }



        function addToCart(index) {

            cart.push(products[index]);

            updateCartUI();

        }



        function removeFromCart(cartIndex) {

            cart.splice(cartIndex, 1);

            updateCartUI();

            renderCartModalItems();

        }



        function updateCartUI() {

            const cartBar = document.getElementById('cartBar');

            const cartCount = document.getElementById('cartCount');

            const cartTotal = document.getElementById('cartTotal');



            cartCount.innerText = cart.length;



            let total = 0;

            cart.forEach(item => { total += parseFloat(item.price); });

            cartTotal.innerText = `R$ ${total.toFixed(2)}`;



            // Se o painel admin estiver aberto, o carrinho NUNCA aparece

            if (cart.length > 0 && !isAdminVisible) {

                cartBar.classList.remove('hidden');

            } else {

                cartBar.classList.add('hidden');

                if (cart.length === 0) closeCartModal();

            }

        }



        function openCartModal() {

            renderCartModalItems();

            document.getElementById('cartModal').classList.remove('hidden');

        }



        function closeCartModal() {

            document.getElementById('cartModal').classList.add('hidden');

        }



        function renderCartModalItems() {

            const container = document.getElementById('cartModalItems');

            const modalTotal = document.getElementById('modalTotal');

            

            if (cart.length === 0) {

                container.innerHTML = '<p class="text-gray-500 italic py-4 text-center">O carrinho está vazio.</p>';

                modalTotal.innerText = 'R$ 0,00';

                return;

            }



            let html = '';

            let total = 0;

            cart.forEach((item, index) => {

                total += parseFloat(item.price);

                html += `

                    <div class="py-3 flex justify-between items-center">

                        <div>

                            <h4 class="font-bold text-sm text-gray-800">${item.name}</h4>

                            <span class="text-green-600 font-semibold text-xs">R$ ${item.price}</span>

                        </div>

                        <button onclick="removeFromCart(${index})" class="text-red-500 hover:text-red-700 text-xs bg-red-50 px-2.5 py-1 rounded font-medium">Remover</button>

                    </div>

                `;

            });

            container.innerHTML = html;

            modalTotal.innerText = `R$ ${total.toFixed(2)}`;

        }



        function checkoutWhatsApp() {

            if (cart.length === 0) return;



            const phone = storeData.phone.replace(/\D/g, '');

            const payment = document.getElementById('paymentMethod').value;

            

            let message = `*Olá! Gostaria de fazer o seguinte pedido:*\n\n`;

            let total = 0;



            cart.forEach((item, idx) => {

                message += `${idx + 1}. ${item.name} - R$ ${item.price}\n`;

                total += parseFloat(item.price);

            });



            message += `\n*Total dos Produtos:* R$ ${total.toFixed(2)}`;

            message += `\n*Forma de Pagamento:* ${payment}`;

            message += `\n\nPor favor, confirmam o pedido e o tempo de entrega?`;



            const encodedMessage = encodeURIComponent(message);

            window.open(`https://wa.me/${phone}?text=${encodedMessage}`, '_blank');

        }



        function saveData() {

            localStorage.setItem('storeData', JSON.stringify(storeData));

            localStorage.setItem('categories', JSON.stringify(categories));

            localStorage.setItem('products', JSON.stringify(products));

        }



        function renderCatalog() {

            const catalogList = document.getElementById('catalogList');

            if (products.length === 0) {

                catalogList.innerHTML = '<p class="text-gray-500 italic">Nenhum produto cadastrado ainda.</p>';

                return;

            }



            let html = '';

            products.forEach((prod, index) => {

                html += `

                    <div class="flex border rounded-lg overflow-hidden shadow-sm bg-white p-3 gap-4 items-center border-gray-200">

                        <img src="${prod.image}" alt="${prod.name}" class="w-20 h-20 object-cover rounded-md border flex-shrink-0">

                        <div class="flex-1">

                            <span class="text-xs bg-gray-100 text-gray-800 px-2 py-0.5 rounded-full font-semibold">${prod.category}</span>

                            <h4 class="font-bold text-lg mt-1 text-gray-800">${prod.name}</h4>

                            <p class="text-green-600 font-semibold">R$ ${prod.price}</p>

                        </div>

                        <div class="flex flex-col gap-2">

                            <!-- O botão de adicionar ao carrinho SÓ APARECE se NÃO estivermos no painel de administração -->

                            ${!isAdminVisible ? `<button onclick="addToCart(${index})" class="bg-blue-600 hover:bg-blue-700 text-white text-xs px-3 py-2 rounded font-bold shadow">🛒 Adicionar</button>` : ''}

                            

                            <!-- O botão de excluir só aparece SE estivermos no painel do administrador -->

                            ${isAdminVisible ? `<button onclick="removeProduct(${index})" class="text-red-500 text-xs px-2 py-1 bg-red-50 rounded">Excluir</button>` : ''}

                        </div>

                    </div>

                `;

            });

            catalogList.innerHTML = html;

        }

    </script>
